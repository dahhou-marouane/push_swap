#ifndef PUSH_SWAP_H

# define PUSH_SWAP_H

# include <unistd.h>

# include <stdlib.h>

# include <limits.h>

typedef struct s_node
{
int             value;
int             index;
struct s_node   *next;
}   t_node;

typedef struct s_stack
{
t_node  *top;
int     size;
}   t_stack;

/* Initialization */
t_stack *init_stack(void);
void    free_stack(t_stack *stack);
void    error_exit(void);

/* Parsing */
int     parse_arguments(int argc, char **argv, t_stack *a);
int     ft_atoi_check(const char *str, int *num);
int     is_duplicate(t_stack *stack, int num);
void    assign_indices(t_stack *stack);

/* Stack Operations */
void    push(t_stack *stack, int value);
int     pop(t_stack *stack);
t_node  *peek(t_stack *stack);

/* Operations - Push */
void    pa(t_stack *a, t_stack *b);
void    pb(t_stack *a, t_stack *b);

/* Operations - Swap */
void    sa(t_stack *a);
void    sb(t_stack *b);
void    ss(t_stack *a, t_stack *b);

/* Operations - Rotate */
void    ra(t_stack *a);
void    rb(t_stack *b);
void    rr(t_stack *a, t_stack *b);

/* Operations - Reverse Rotate */
void    rra(t_stack *a);
void    rrb(t_stack *b);
void    rrr(t_stack *a, t_stack *b);

/* Sorting Algorithms */
void    sort_three(t_stack *a);
void    sort_small(t_stack *a, t_stack *b);
void    sort_chunk(t_stack *a, t_stack *b);

/* Utils */
int     is_sorted(t_stack *stack);
int     find_min(t_stack *stack);
int     find_max(t_stack *stack);
int     get_position(t_stack *stack, int index);
int     ft_strlen(const char *s);

#endif

int ft_strlen(const char *s)
{
int i = 0;
while (s[i])
i++;
return (i);
}

void error_exit(void)
{
write(2, “Error\n”, 6);
exit(1);
}

int is_sorted(t_stack *stack)
{
t_node *current;

```
if (!stack || !stack->top)
    return (1);
current = stack->top;
while (current->next)
{
    if (current->index > current->next->index)
        return (0);
    current = current->next;
}
return (1);
```

}

int find_min(t_stack *stack)
{
t_node  *current;
int     min;

```
if (!stack || !stack->top)
    return (-1);
current = stack->top;
min = current->index;
while (current)
{
    if (current->index < min)
        min = current->index;
    current = current->next;
}
return (min);
```

}

int find_max(t_stack *stack)
{
t_node  *current;
int     max;

```
if (!stack || !stack->top)
    return (-1);
current = stack->top;
max = current->index;
while (current)
{
    if (current->index > max)
        max = current->index;
    current = current->next;
}
return (max);
```

}

int get_position(t_stack *stack, int index)
{
t_node  *current;
int     pos;

```
if (!stack || !stack->top)
    return (-1);
current = stack->top;
pos = 0;
while (current)
{
    if (current->index == index)
        return (pos);
    pos++;
    current = current->next;
}
return (-1);
```

}

t_stack *init_stack(void)
{
t_stack *stack;

```
stack = malloc(sizeof(t_stack));
if (!stack)
    return (NULL);
stack->top = NULL;
stack->size = 0;
return (stack);
```

}

void free_stack(t_stack *stack)
{
t_node *current;
t_node *next;

```
if (!stack)
    return;
current = stack->top;
while (current)
{
    next = current->next;
    free(current);
    current = next;
}
free(stack);
```

}

void push(t_stack *stack, int value)
{
t_node *new_node;

```
new_node = malloc(sizeof(t_node));
if (!new_node)
    error_exit();
new_node->value = value;
new_node->index = -1;
new_node->next = stack->top;
stack->top = new_node;
stack->size++;
```

}

int pop(t_stack *stack)
{
t_node  *temp;
int     value;

```
if (!stack || !stack->top)
    return (0);
temp = stack->top;
value = temp->value;
stack->top = temp->next;
free(temp);
stack->size--;
return (value);
```

}

t_node *peek(t_stack *stack)
{
if (!stack)
return (NULL);
return (stack->top);
}

int ft_atoi_check(const char *str, int *num)
{
long    result;
int     sign;
int     i;

```
result = 0;
sign = 1;
i = 0;
if (str[i] == '-' || str[i] == '+')
{
    if (str[i] == '-')
        sign = -1;
    i++;
}
if (!str[i])
    return (0);
while (str[i])
{
    if (str[i] < '0' || str[i] > '9')
        return (0);
    result = result * 10 + (str[i] - '0');
    if (result * sign > INT_MAX || result * sign < INT_MIN)
        return (0);
    i++;
}
*num = (int)(result * sign);
return (1);
```

}

int is_duplicate(t_stack *stack, int num)
{
t_node *current;

```
current = stack->top;
while (current)
{
    if (current->value == num)
        return (1);
    current = current->next;
}
return (0);
```

}

void assign_indices(t_stack *stack)
{
t_node  *current;
t_node  *compare;
int     index;

```
current = stack->top;
while (current)
{
    index = 0;
    compare = stack->top;
    while (compare)
    {
        if (compare->value < current->value)
            index++;
        compare = compare->next;
    }
    current->index = index;
    current = current->next;
}
```

}

int parse_arguments(int argc, char **argv, t_stack *a)
{
int i;
int num;

```
i = argc - 1;
while (i >= 1)
{
    if (!ft_atoi_check(argv[i], &num))
        return (0);
    if (is_duplicate(a, num))
        return (0);
    push(a, num);
    i--;
}
assign_indices(a);
return (1);
```

}

void sort_three(t_stack *a)
{
int first;
int second;
int third;

```
first = a->top->index;
second = a->top->next->index;
third = a->top->next->next->index;

if (first > second && second < third && first < third)
    sa(a);
else if (first > second && second > third)
{
    sa(a);
    rra(a);
}
else if (first > second && second < third && first > third)
    ra(a);
else if (first < second && second > third && first < third)
{
    sa(a);
    ra(a);
}
else if (first < second && second > third && first > third)
    rra(a);
```

}

void sort_small(t_stack *a, t_stack *b)
{
int min_pos;

```
if (a->size == 2)
{
    if (a->top->index > a->top->next->index)
        sa(a);
    return;
}
if (a->size == 3)
{
    sort_three(a);
    return;
}
while (a->size > 3)
{
    min_pos = get_position(a, find_min(a));
    if (min_pos <= a->size / 2)
    {
        while (min_pos-- > 0)
            ra(a);
    }
    else
    {
        while (min_pos++ < a->size)
            rra(a);
    }
    pb(a, b);
}
sort_three(a);
while (b->size > 0)
    pa(a, b);
```

}

static int calculate_chunk_size(int size)
{
if (size <= 100)
return (15);
else if (size <= 500)
return (30);
else
return (45);
}

static void push_chunk_to_b(t_stack *a, t_stack *b, int chunk_min, int chunk_max)
{
int size;
int i;

```
size = a->size;
i = 0;
while (i < size)
{
    if (a->top->index >= chunk_min && a->top->index <= chunk_max)
    {
        pb(a, b);
        if (b->top->index < chunk_min + (chunk_max - chunk_min) / 2)
            rb(b);
    }
    else
        ra(a);
    i++;
}
```

}

static void push_max_to_a(t_stack *a, t_stack *b)
{
int max;
int max_pos;

```
max = find_max(b);
max_pos = get_position(b, max);

if (max_pos <= b->size / 2)
{
    while (b->top->index != max)
        rb(b);
}
else
{
    while (b->top->index != max)
        rrb(b);
}
pa(a, b);
```

}

void sort_chunk(t_stack *a, t_stack *b)
{
int chunk_size;
int chunk_min;
int chunk_max;
int total_size;

```
total_size = a->size;
chunk_size = calculate_chunk_size(total_size);
chunk_min = 0;
chunk_max = chunk_size - 1;

while (a->size > 0)
{
    push_chunk_to_b(a, b, chunk_min, chunk_max);
    chunk_min += chunk_size;
    chunk_max += chunk_size;
    if (chunk_max >= total_size)
        chunk_max = total_size - 1;
}

while (b->size > 0)
    push_max_to_a(a, b);
```

}

int main(int argc, char **argv)
{
t_stack *a;
t_stack *b;

```
if (argc < 2)
    return (0);

a = init_stack();
b = init_stack();

if (!a || !b)
{
    free_stack(a);
    free_stack(b);
    error_exit();
}

if (!parse_arguments(argc, argv, a))
{
    free_stack(a);
    free_stack(b);
    error_exit();
}

if (is_sorted(a))
{
    free_stack(a);
    free_stack(b);
    return (0);
}

if (a->size <= 5)
    sort_small(a, b);
else
    sort_chunk(a, b);

free_stack(a);
free_stack(b);
return (0);
```
